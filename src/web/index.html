SYNTIQ grammar (context-free, EBNF) 
program      ::= blueprint
blueprint    ::= "blueprint" IDENT "{" { method } "}"
method       ::= { modifier } type ( IDENT | "start" ) "(" [ param { "," param } ] ")" block
param        ::= type IDENT
modifier     ::= "open" | "hidden" | "guarded" | "shared" | "locked"
type         ::= "whole" | "bigwhole" | "smallwhole" | "tinywhole" | "decimal"
               | "shortdecimal" | "letter" | "truth" | "text" | "empty"

block        ::= "{" { statement } "}"
statement    ::= declaration ";"
               | assignment ";"
               | "show" "(" expr ")" ";"
               | ifStmt | whileStmt | forStmt
               | "stop" ";" | "skip" ";" | "giveback" [ expr ] ";"
               | block
declaration  ::= type IDENT [ ":=" expr ]
assignment   ::= IDENT ":=" expr
ifStmt       ::= "check" "(" expr ")" block [ "otherwise" ( ifStmt | block ) ]
whileStmt    ::= "repeat" "(" expr ")" block
forStmt      ::= "cycle" "(" ( declaration | assignment ) ";" expr ";" assignment ")" block

expr         ::= andExpr { "either" andExpr }
andExpr      ::= equality { "both" equality }
equality     ::= relational { ( "~=" | "~!" ) relational }
relational   ::= additive { ( "~>" | "~<" ) additive }
additive     ::= term { ( "~+" | "~-" ) term }
term         ::= unary { ( "~*" | "~/" | "~%" ) unary }
unary        ::= ( "negate" | "~-" ) unary | primary
primary      ::= INTEGER | DECIMAL | STRING | CHAR | "valid" | "invalid" | "nil"
               | IDENT | "(" expr ")"

SYNTIQ Grammar (Context-Free, EBNF)


    
program      ::= blueprint
blueprint    ::= "blueprint" IDENT "{" { method } "}"

method       ::= { modifier } type ( IDENT | "start" )
                 "(" [ param { "," param } ] ")" block

param        ::= type IDENT

modifier     ::= "open" | "hidden" | "guarded" | "shared" | "locked"

type         ::= "whole" | "bigwhole" | "smallwhole" | "tinywhole"
               | "decimal" | "shortdecimal" | "letter"
               | "truth" | "text" | "empty"

block        ::= "{" { statement } "}"

statement    ::= declaration ";"
               | assignment ";"
               | "show" "(" expr ")" ";"
               | ifStmt | whileStmt | forStmt
               | "stop" ";" | "skip" ";" | "giveback" [ expr ] ";"
               | block

declaration  ::= type IDENT [ ":=" expr ]
assignment   ::= IDENT ":=" expr

ifStmt       ::= "check" "(" expr ")" block
                 [ "otherwise" ( ifStmt | block ) ]

whileStmt    ::= "repeat" "(" expr ")" block

forStmt      ::= "cycle" "(" ( declaration | assignment )
                 ";" expr ";" assignment ")" block

expr         ::= andExpr { "either" andExpr }
andExpr      ::= equality { "both" equality }

equality     ::= relational { ( "~=" | "~!" ) relational }
relational   ::= additive { ( "~>" | "~<" ) additive }
additive     ::= term { ( "~+" | "~-" ) term }
term         ::= unary { ( "~*" | "~/" | "~%" ) unary }

unary        ::= ( "negate" | "~-" ) unary | primary

primary      ::= INTEGER | DECIMAL | STRING | CHAR
               | "valid" | "invalid" | "nil"
               | call | IDENT | "(" expr ")"

call         ::= IDENT "(" [ arguments ] ")"

arguments    ::= expr { "," expr }
